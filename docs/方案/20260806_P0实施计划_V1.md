```ts
// src/core/model/track.ts
export type TrackFormat = 'gpx' | 'kml' | 'fit'
export type Crs = 'wgs84' | 'gcj02' | 'bd09'

export interface TrackPoints {
  lon: Float64Array
  lat: Float64Array
  ele?: Float32Array          // 米,缺失整列 undefined,单点缺失 NaN
  time?: Float64Array         // epoch ms,同上
  hr?: Uint16Array
  /** 累计里程(米),由 geo 模块计算后挂载 */
  cumDist?: Float64Array
}

export interface TrackMeta { name: string; format: TrackFormat; fileName: string; creator?: string }

export interface Track {
  id: string
  meta: TrackMeta
  crs: 'wgs84'                // 内部永远 WGS-84
  originalCrs: Crs            // 导入时识别到的原始坐标系
  points: TrackPoints
}

export interface TrackPointsInput {
  lon: ArrayLike<number>; lat: ArrayLike<number>
  ele?: ArrayLike<number>; time?: ArrayLike<number>; hr?: ArrayLike<number>
}

let seq = 0
export function newId(prefix: string): string {
  return `${prefix}_${Date.now().toString(36)}_${(seq++).toString(36)}`
}

export function createTrack(pts: TrackPointsInput, meta: TrackMeta, originalCrs: Crs = 'wgs84'): Track {
  const n = pts.lon.length
  if (pts.lat.length !== n || (pts.ele && pts.ele.length !== n) || (pts.time && pts.time.length !== n))
    throw new Error(`point array length mismatch`)
  return {
    id: newId('trk'),
    meta, crs: 'wgs84', originalCrs,
    points: {
      lon: Float64Array.from(pts.lon), lat: Float64Array.from(pts.lat),
      ele: pts.ele ? Float32Array.from(pts.ele) : undefined,
      time: pts.time ? Float64Array.from(pts.time) : undefined,
      hr: pts.hr ? Uint16Array.from(pts.hr) : undefined,
    },
  }
}

export function trackPointCount(t: Track): number { return t.points.lon.length }
```

```ts
// src/core/geo/distance.ts
const R = 6371008.8
const rad = Math.PI / 180

export function haversine(lon1: number, lat1: number, lon2: number, lat2: number): number {
  const dLat = (lat2 - lat1) * rad, dLon = (lon2 - lon1) * rad
  const a = Math.sin(dLat / 2) ** 2 +
    Math.cos(lat1 * rad) * Math.cos(lat2 * rad) * Math.sin(dLon / 2) ** 2
  return 2 * R * Math.asin(Math.sqrt(a))
}

/** 累计里程数组,cum[0]=0,单位米 */
export function computeCumDist(lon: Float64Array, lat: Float64Array): Float64Array {
  const n = lon.length
  const cum = new Float64Array(n)
  for (let i = 1; i < n; i++) cum[i] = cum[i - 1] + haversine(lon[i - 1], lat[i - 1], lon[i], lat[i])
  return cum
}

/** 返回 dist 所在段的起点索引(二分),dist 超界时夹到边界段 */
export function locateByDist(cum: Float64Array, dist: number): number {
  let lo = 0, hi = cum.length - 1
  if (dist <= 0) return 0
  if (dist >= cum[hi]) return hi - 1
  while (hi - lo > 1) { const mid = (lo + hi) >> 1; cum[mid] <= dist ? (lo = mid) : (hi = mid) }
  return lo
}
```

性能关键:**不用 DOM/fast-xml-parser 全量建树解析 105MB 文件**,`<trkpt>` 主循环用正则扫描(GPX 结构高度规整),元数据(creator/name)用 fast-xml-parser 只解析文件头 4KB。

```ts
// src/core/parsers/gpx.ts
import { XMLParser } from 'fast-xml-parser'
import { createTrack, type Track } from '../model/track'

const TRKPT_RE = /<trkpt\b[^>]*\blat="(-?[\d.]+)"[^>]*\blon="(-?[\d.]+)"[^>]*>([\s\S]*?)<\/trkpt>|<trkpt\b[^>]*\blon="(-?[\d.]+)"[^>]*\blat="(-?[\d.]+)"[^>]*\/>/g
const ELE_RE = /<ele>(-?[\d.]+)<\/ele>/
const TIME_RE = /<time>([^<]+)<\/time>/
const HR_RE = /<(?:\w+:)?hr>(\d+)<\/(?:\w+:)?hr>/

export function parseGpx(xml: string, fileName: string): Track {
  // 1) 元数据:只喂头部,避免全量建树
  const head = xml.slice(0, 4096)
  const creator = /creator="([^"]*)"/.exec(head)?.[1]
  const name = /<name>([^<]*)<\/name>/.exec(xml.slice(0, 65536))?.[1]?.trim()

  // 2) 主循环:正则扫描 trkpt
  const lon: number[] = [], lat: number[] = [], ele: number[] = [], time: number[] = [], hr: number[] = []
  let hasEle = false, hasTime = false, hasHr = false
  // 统一用带内容的形式匹配(lat/lon 属性顺序两种都支持)
  const re = /<trkpt\b([^>]*)>([\s\S]*?)<\/trkpt>/g
  let m: RegExpExecArray | null
  while ((m = re.exec(xml))) {
    const attrs = m[1], body = m[2]
    const la = /\blat="(-?[\d.]+)"/.exec(attrs), lo = /\blon="(-?[\d.]+)"/.exec(attrs)
    if (!la || !lo) continue
    lat.push(+la[1]); lon.push(+lo[1])
    const e = ELE_RE.exec(body); if (e) { hasEle = true; ele.push(+e[1]) } else ele.push(NaN)
    const t = TIME_RE.exec(body); if (t) { hasTime = true; time.push(Date.parse(t[1])) } else time.push(NaN)
    const h = HR_RE.exec(body); if (h) { hasHr = true; hr.push(+h[1]) } else hr.push(0)
  }
  if (lon.length === 0) throw new Error(`GPX 无轨迹点: ${fileName}`)
  return createTrack(
    { lon, lat, ele: hasEle ? ele : undefined, time: hasTime ? time : undefined, hr: hasHr ? hr : undefined },
    { name: name ?? fileName, format: 'gpx', fileName, creator },
  )
}
```

```ts
// src/core/parsers/kml.ts
import { createTrack, type Track } from '../model/track'

export function parseKml(xml: string, fileName: string): Track {
  const name = /<name>([^<]*)<\/name>/.exec(xml)?.[1]?.trim()
  const lon: number[] = [], lat: number[] = [], ele: number[] = [], time: number[] = []
  let hasEle = false, hasTime = false

  // gx:Track 优先(带时间)
  const coords = [...xml.matchAll(/<gx:coord>([^<]+)<\/gx:coord>/g)]
  if (coords.length > 0) {
    const whens = [...xml.matchAll(/<when>([^<]+)<\/when>/g)]
    coords.forEach((c, i) => {
      const [x, y, z] = c[1].trim().split(/\s+/).map(Number)
      lon.push(x); lat.push(y)
      if (Number.isFinite(z)) { hasEle = true; ele.push(z) } else ele.push(NaN)
      const w = whens[i]?.[1]
      if (w) { hasTime = true; time.push(Date.parse(w)) } else time.push(NaN)
    })
  } else {
    // 所有 LineString 的 coordinates 顺序拼接
    for (const m of xml.matchAll(/<coordinates>([\s\S]*?)<\/coordinates>/g)) {
      for (const tuple of m[1].trim().split(/\s+/)) {
        const parts = tuple.split(',').map(Number)
        if (parts.length < 2 || !Number.isFinite(parts[0])) continue
        lon.push(parts[0]); lat.push(parts[1])
        if (parts.length >= 3 && Number.isFinite(parts[2])) { hasEle = true; ele.push(parts[2]) } else ele.push(NaN)
        time.push(NaN)
      }
    }
  }
  if (lon.length === 0) throw new Error(`KML 无轨迹点: ${fileName}`)
  return createTrack(
    { lon, lat, ele: hasEle ? ele : undefined, time: hasTime ? time : undefined },
    { name: name ?? fileName, format: 'kml', fileName },
  )
}
```

```ts
// src/core/parsers/fit.ts
import FitParser from 'fit-file-parser'
import { createTrack, type Track } from '../model/track'

interface FitRecord {
  position_lat?: number; position_long?: number; altitude?: number
  timestamp?: Date; heart_rate?: number
}

export function recordsToTrack(records: FitRecord[], fileName: string): Track {
  const lon: number[] = [], lat: number[] = [], ele: number[] = [], time: number[] = [], hr: number[] = []
  let hasEle = false, hasHr = false
  for (const r of records) {
    if (typeof r.position_lat !== 'number' || typeof r.position_long !== 'number') continue
    lat.push(r.position_lat); lon.push(r.position_long)
    if (typeof r.altitude === 'number') { hasEle = true; ele.push(r.altitude) } else ele.push(NaN)
    time.push(r.timestamp ? r.timestamp.getTime() : NaN)
    if (typeof r.heart_rate === 'number') { hasHr = true; hr.push(r.heart_rate) } else hr.push(0)
  }
  if (lon.length === 0) throw new Error(`FIT 无轨迹点: ${fileName}`)
  return createTrack(
    { lon, lat, ele: hasEle ? ele : undefined, time, hr: hasHr ? hr : undefined },
    { name: fileName.replace(/\.fit$/i, ''), format: 'fit', fileName },
  )
}

export function parseFit(buf: ArrayBuffer, fileName: string): Promise<Track> {
  return new Promise((resolve, reject) => {
    const p = new FitParser({ force: true, speedUnit: 'm/s', lengthUnit: 'm', elevationValues: 'absolute' })
    p.parse(buf, (err: unknown, data: { records?: FitRecord[] }) => {
      if (err) return reject(err instanceof Error ? err : new Error(String(err)))
      try { resolve(recordsToTrack(data.records ?? [], fileName)) } catch (e) { reject(e) }
    })
  })
}
```

```ts
// src/core/crs/transform.ts
// GCJ-02 偏转算法为社区公开近似实现(参见 wandergis/coordtransform, MIT)
const a = 6378245.0
const ee = 0.00669342162296594323
const xPI = (Math.PI * 3000.0) / 180.0

export function outOfChina(lon: number, lat: number): boolean {
  return !(lon > 73.66 && lon < 135.05 && lat > 3.86 && lat < 53.55)
}

function tLat(x: number, y: number): number {
  let r = -100 + 2 * x + 3 * y + 0.2 * y * y + 0.1 * x * y + 0.2 * Math.sqrt(Math.abs(x))
  r += ((20 * Math.sin(6 * x * Math.PI) + 20 * Math.sin(2 * x * Math.PI)) * 2) / 3
  r += ((20 * Math.sin(y * Math.PI) + 40 * Math.sin((y / 3) * Math.PI)) * 2) / 3
  r += ((160 * Math.sin((y / 12) * Math.PI) + 320 * Math.sin((y * Math.PI) / 30)) * 2) / 3
  return r
}
function tLon(x: number, y: number): number {
  let r = 300 + x + 2 * y + 0.1 * x * x + 0.1 * x * y + 0.1 * Math.sqrt(Math.abs(x))
  r += ((20 * Math.sin(6 * x * Math.PI) + 20 * Math.sin(2 * x * Math.PI)) * 2) / 3
  r += ((20 * Math.sin(x * Math.PI) + 40 * Math.sin((x / 3) * Math.PI)) * 2) / 3
  r += ((150 * Math.sin((x / 12) * Math.PI) + 300 * Math.sin((x / 30) * Math.PI)) * 2) / 3
  return r
}

export function wgs84ToGcj02(lon: number, lat: number): [number, number] {
  if (outOfChina(lon, lat)) return [lon, lat]
  let dLat = tLat(lon - 105.0, lat - 35.0)
  let dLon = tLon(lon - 105.0, lat - 35.0)
  const radLat = (lat / 180.0) * Math.PI
  let magic = Math.sin(radLat); magic = 1 - ee * magic * magic
  const sqrtMagic = Math.sqrt(magic)
  dLat = (dLat * 180.0) / (((a * (1 - ee)) / (magic * sqrtMagic)) * Math.PI)
  dLon = (dLon * 180.0) / ((a / sqrtMagic) * Math.cos(radLat) * Math.PI)
  return [lon + dLon, lat + dLat]
}

/** 迭代精确反解,残差 < 0.5m */
export function gcj02ToWgs84(lon: number, lat: number): [number, number] {
  if (outOfChina(lon, lat)) return [lon, lat]
  let wlon = lon, wlat = lat
  for (let i = 0; i < 3; i++) {
    const [glon, glat] = wgs84ToGcj02(wlon, wlat)
    wlon -= glon - lon; wlat -= glat - lat
  }
  return [wlon, wlat]
}

export function gcj02ToBd09(lon: number, lat: number): [number, number] {
  const z = Math.sqrt(lon * lon + lat * lat) + 0.00002 * Math.sin(lat * xPI)
  const theta = Math.atan2(lat, lon) + 0.000003 * Math.cos(lon * xPI)
  return [z * Math.cos(theta) + 0.0065, z * Math.sin(theta) + 0.006]
}

export function bd09ToGcj02(lon: number, lat: number): [number, number] {
  const x = lon - 0.0065, y = lat - 0.006
  const z = Math.sqrt(x * x + y * y) - 0.00002 * Math.sin(y * xPI)
  const theta = Math.atan2(y, x) - 0.000003 * Math.cos(x * xPI)
  return [z * Math.cos(theta), z * Math.sin(theta)]
}

import type { Crs } from '../model/track'

const pointFns: Record<string, (lon: number, lat: number) => [number, number]> = {
  'gcj02>wgs84': gcj02ToWgs84, 'wgs84>gcj02': wgs84ToGcj02,
  'bd09>gcj02': bd09ToGcj02, 'gcj02>bd09': gcj02ToBd09,
  'bd09>wgs84': (lo, la) => { const [g0, g1] = bd09ToGcj02(lo, la); return gcj02ToWgs84(g0, g1) },
  'wgs84>bd09': (lo, la) => { const [g0, g1] = wgs84ToGcj02(lo, la); return gcj02ToBd09(g0, g1) },
}

export function convertTrackArrays(lon: Float64Array, lat: Float64Array, from: Crs, to: Crs) {
  if (from === to) return { lon: Float64Array.from(lon), lat: Float64Array.from(lat) }
  const fn = pointFns[`${from}>${to}`]
  if (!fn) throw new Error(`unsupported crs conversion ${from}>${to}`)
  const n = lon.length, outLon = new Float64Array(n), outLat = new Float64Array(n)
  for (let i = 0; i < n; i++) { const [x, y] = fn(lon[i], lat[i]); outLon[i] = x; outLat[i] = y }
  return { lon: outLon, lat: outLat }
}
```

```ts
// src/core/crs/detect.ts
import type { Crs } from '../model/track'

export interface DetectInput { creator?: string; fileName?: string }
export interface DetectResult { crs: Crs; confidence: 'high' | 'unknown'; reason: string }

const WGS84_PAT = /coros|garmin|suunto|strava|polar|wahoo|amazfit|huami|apple|gpx.?studio|caltopo|komoot/i
const GCJ02_PAT = /两步路|foooooot|六只脚|lvye|行者|imxingzhe|xingzhe|奥维|ovital|高德|amap|keep/i
const BD09_PAT = /百度|baidu/i

export function detectCrs(input: DetectInput, sourceMemory: Record<string, Crs>): DetectResult {
  const sig = `${input.creator ?? ''} ${input.fileName ?? ''}`
  if (input.creator && sourceMemory[input.creator])
    return { crs: sourceMemory[input.creator], confidence: 'high', reason: 'remembered source' }
  if (BD09_PAT.test(sig)) return { crs: 'bd09', confidence: 'high', reason: 'creator/filename matches baidu' }
  if (GCJ02_PAT.test(sig)) return { crs: 'gcj02', confidence: 'high', reason: 'creator/filename matches cn-app' }
  if (WGS84_PAT.test(sig)) return { crs: 'wgs84', confidence: 'high', reason: 'creator/filename matches device' }
  return { crs: 'wgs84', confidence: 'unknown', reason: 'no signal; require user confirm' }
}
```

```ts
// src/core/pipeline/import.ts —— 统一导入管线(worker 内调用)
import { parseGpx } from '../parsers/gpx'
import { parseKml } from '../parsers/kml'
import { parseFit } from '../parsers/fit'
import { detectCrs, type DetectResult } from '../crs/detect'
import { convertTrackArrays } from '../crs/transform'
import { computeCumDist } from '../geo/distance'
import type { Crs, Track } from '../model/track'

export interface ImportResult { track: Track; detect: DetectResult }

export async function importFile(
  fileName: string, data: string | ArrayBuffer,
  sourceMemory: Record<string, Crs>, forcedCrs?: Crs,
): Promise<ImportResult> {
  const ext = fileName.toLowerCase().split('.').pop()
  let track: Track
  if (ext === 'gpx') track = parseGpx(data as string, fileName)
  else if (ext === 'kml') track = parseKml(data as string, fileName)
  else if (ext === 'fit') track = await parseFit(data as ArrayBuffer, fileName)
  else throw new Error(`不支持的格式: ${ext}`)

  const detect = forcedCrs
    ? { crs: forcedCrs, confidence: 'high' as const, reason: 'user forced' }
    : detectCrs({ creator: track.meta.creator, fileName }, sourceMemory)

  if (detect.crs !== 'wgs84') {
    const { lon, lat } = convertTrackArrays(track.points.lon, track.points.lat, detect.crs, 'wgs84')
    track = { ...track, originalCrs: detect.crs, points: { ...track.points, lon, lat } }
  }
  track.points.cumDist = computeCumDist(track.points.lon, track.points.lat)
  return { track, detect }
}
```

```ts
/** 均匀抽稀索引:返回长度 ≤ maxPoints 的递增索引数组,首末点必含 */
export function decimateIndices(n: number, maxPoints: number): Uint32Array {
  if (n <= maxPoints) return Uint32Array.from({ length: n }, (_, i) => i)
  const out = new Uint32Array(maxPoints)
  const step = (n - 1) / (maxPoints - 1)
  for (let i = 0; i < maxPoints; i++) out[i] = Math.round(i * step)
  out[maxPoints - 1] = n - 1
  return out
}
```

```ts
import type { Track } from '../core/model/track'
import { decimateIndices } from '../core/toolbox/decimate'
import { locateByDist } from '../core/geo/distance'

export interface ProfileData {
  dist: Float64Array; ele: Float32Array; idx: Uint32Array   // 抽稀后
  totalDist: number; minEle: number; maxEle: number
}

export function buildProfileData(t: Track, maxPoints = 2000): ProfileData | undefined {
  if (!t.points.ele || !t.points.cumDist) return undefined
  const idx = decimateIndices(t.points.lon.length, maxPoints)
  const dist = Float64Array.from(idx, (i) => t.points.cumDist![i])
  const ele = Float32Array.from(idx, (i) => t.points.ele![i])
  let minEle = Infinity, maxEle = -Infinity
  for (const e of ele) { if (Number.isFinite(e)) { if (e < minEle) minEle = e; if (e > maxEle) maxEle = e } }
  return { dist, ele, idx, totalDist: dist[dist.length - 1], minEle, maxEle }
}

/** 像素 x → 全精度点索引 */
export function xToIndex(p: ProfileData, x: number, width: number): number {
  const d = (x / width) * p.totalDist
  return p.idx[locateByDist(p.dist, d)]
}

  const d = t.points.cumDist![index]
  return (d / p.totalDist) * width
}
```

```ts
// src/core/toolbox/ops.ts
import { createTrack, type Track, type TrackPoints } from '../model/track'
import { computeCumDist, haversine } from '../geo/distance'

function slicePoints(p: TrackPoints, keep: ArrayLike<number>): Parameters<typeof createTrack>[0] {
  const pick = (arr?: { [i: number]: number }) => arr ? Array.from(keep, (i) => (arr as any)[i] as number) : undefined
  return { lon: pick(p.lon)!, lat: pick(p.lat)!, ele: pick(p.ele), time: pick(p.time), hr: pick(p.hr) }
}

function derive(src: Track, keep: ArrayLike<number>, suffix: string): Track {
  const t = createTrack(slicePoints(src.points, keep), { ...src.meta, name: `${src.meta.name}${suffix}` }, src.originalCrs)
  t.points.cumDist = computeCumDist(t.points.lon, t.points.lat)
  return t
}

export function splitAt(t: Track, index: number): [Track, Track] {
  const n = t.points.lon.length
  if (index <= 0 || index >= n - 1) throw new Error('split index out of range')
  const a = Array.from({ length: index + 1 }, (_, i) => i)
  const b = Array.from({ length: n - index }, (_, i) => i + index)
  return [derive(t, a, ' ·A'), derive(t, b, ' ·B')]
}

export function joinTracks(list: Track[]): Track {
  if (list.length < 2) throw new Error('need >= 2 tracks')
  const merged = { lon: [] as number[], lat: [] as number[], ele: [] as number[], time: [] as number[] }
  let hasEle = false, hasTime = false
  for (const t of list) for (let i = 0; i < t.points.lon.length; i++) {
    merged.lon.push(t.points.lon[i]); merged.lat.push(t.points.lat[i])
    const e = t.points.ele?.[i]; if (e !== undefined && Number.isFinite(e)) { hasEle = true; merged.ele.push(e) } else merged.ele.push(NaN)
    const tm = t.points.time?.[i]; if (tm !== undefined && Number.isFinite(tm)) { hasTime = true; merged.time.push(tm) } else merged.time.push(NaN)
  }
  const out = createTrack(
    { lon: merged.lon, lat: merged.lat, ele: hasEle ? merged.ele : undefined, time: hasTime ? merged.time : undefined },
    { ...list[0].meta, name: `${list[0].meta.name} ·拼接` }, list[0].originalCrs)
  out.points.cumDist = computeCumDist(out.points.lon, out.points.lat)
  return out
}

export function reverseTrack(t: Track): Track {
  const n = t.points.lon.length
  return derive(t, Array.from({ length: n }, (_, i) => n - 1 - i), ' ·反向')
}

export interface AnomalyOptions { maxSpeed?: number /* m/s, default 10 */ }
export function removeAnomalies(t: Track, opt: AnomalyOptions = {}): Track {
  const maxSpeed = opt.maxSpeed ?? 10
  const { lon, lat, time } = t.points
  const keep: number[] = [0]
  for (let i = 1; i < lon.length; i++) {
    const prev = keep[keep.length - 1]
    const d = haversine(lon[prev], lat[prev], lon[i], lat[i])
    const dt = time && Number.isFinite(time[i]) && Number.isFinite(time[prev]) ? (time[i] - time[prev]) / 1000 : NaN
    const v = Number.isFinite(dt) && dt > 0 ? d / dt : 0 // 无时间戳时不按速度剔除
    if (v <= maxSpeed) keep.push(i)
  }
  return derive(t, keep, ' ·清洗')
}

/** Douglas-Peucker,容差单位米(用等矩形近似投影到米平面) */
export function simplifyTrack(t: Track, toleranceM: number): Track {
  const { lon, lat } = t.points
  const n = lon.length
  const latRad = (lat[0] * Math.PI) / 180
  const mx = 111320 * Math.cos(latRad), my = 110540
  const X = (i: number) => lon[i] * mx, Y = (i: number) => lat[i] * my
  const keepFlag = new Uint8Array(n); keepFlag[0] = keepFlag[n - 1] = 1
  const stack: [number, number][] = [[0, n - 1]]
  while (stack.length) {
    const [s, e] = stack.pop()!
    if (e - s < 2) continue
    const x1 = X(s), y1 = Y(s), dx = X(e) - x1, dy = Y(e) - y1
    const len2 = dx * dx + dy * dy
    let maxD = -1, maxI = -1
    for (let i = s + 1; i < e; i++) {
      const px = X(i) - x1, py = Y(i) - y1
      const tproj = len2 === 0 ? 0 : Math.max(0, Math.min(1, (px * dx + py * dy) / len2))
      const ddx = px - tproj * dx, ddy = py - tproj * dy
      const d2 = ddx * ddx + ddy * ddy
      if (d2 > maxD) { maxD = d2; maxI = i }
    }
    if (maxD > toleranceM * toleranceM) { keepFlag[maxI] = 1; stack.push([s, maxI], [maxI, e]) }
  }
  const keep: number[] = []
  for (let i = 0; i < n; i++) if (keepFlag[i]) keep.push(i)
  return derive(t, keep, ' ·抽稀')
}
```

```ts
export class History<T> {
  private undoStack: Snapshot<T>[] = []
  private redoStack: Snapshot<T>[] = []
```

```ts
// src/core/stats/anchor.ts
import { haversine } from '../geo/distance'

/**
 * 里程单调约束锚定:按 CP 输入顺序,每个 CP 只在上一锚点之后的索引里找最近点。
 * 返回与 clicks 等长的锚定索引数组。
 */
export function anchorMonotonic(
  lon: Float64Array, lat: Float64Array, clicks: [number, number][],
): number[] {
  const out: number[] = []
  let from = 0
  for (const [clon, clat] of clicks) {
    let best = from, bestD = Infinity
    for (let i = from; i < lon.length; i++) {
      const d = haversine(clon, clat, lon[i], lat[i])
      if (d < bestD) { bestD = d; best = i }
    }
    out.push(best)
    from = Math.min(best + 1, lon.length - 1)
  }
  return out
}
```

```ts
// src/core/stats/elevation.ts
/** 滑动窗口均值平滑(NaN 不参与也不污染窗口) */
export function smoothElevation(ele: Float32Array, window = 5): Float32Array {
  const n = ele.length, half = window >> 1, out = new Float32Array(n)
  for (let i = 0; i < n; i++) {
    if (Number.isNaN(ele[i])) { out[i] = NaN; continue }
    let sum = 0, cnt = 0
    for (let j = Math.max(0, i - half); j <= Math.min(n - 1, i + half); j++)
      if (!Number.isNaN(ele[j])) { sum += ele[j]; cnt++ }
    out[i] = sum / cnt
  }
  return out
}

/** 阈值滞回累计爬升:仅当相对参考点连续变化超过 threshold 才计入并移动参考点 */
export function computeGainLoss(ele: Float32Array, threshold = 5): { gain: number; loss: number } {
  let gain = 0, loss = 0, ref = NaN
  for (let i = 0; i < ele.length; i++) {
    const e = ele[i]
    if (Number.isNaN(e)) continue
    if (Number.isNaN(ref)) { ref = e; continue }
    if (e - ref >= threshold) { gain += e - ref; ref = e }
    else if (ref - e >= threshold) { loss += ref - e; ref = e }
  }
  return { gain, loss }
}
```

```ts
// src/core/stats/segments.ts
import type { Track } from '../model/track'
import type { CheckPoint } from '../model/checkpoint'
import { smoothElevation, computeGainLoss } from './elevation'

export interface SegmentStats {
  fromName: string; toName: string
  fromIndex: number; toIndex: number
  dist: number            // 米
  gain: number; loss: number
  gainRate: number        // gain/dist
  lossRate: number        // loss/dist
  netSlope: number        // (ele[to]-ele[from])/dist
}

export interface StatsOptions { threshold?: number; smoothWindow?: number }

export function computeSegments(track: Track, cps: CheckPoint[], opt: StatsOptions = {}): SegmentStats[] {
  const { cumDist, ele } = track.points
  if (!cumDist) throw new Error('cumDist required')
  const smoothed = ele ? smoothElevation(ele, opt.smoothWindow ?? 5) : undefined
  const th = opt.threshold ?? 5
  const n = track.points.lon.length
  const bounds = [
    { name: '起点', index: 0 },
    ...cps.map((c) => ({ name: c.name, index: c.anchorIndex })).sort((a, b) => a.index - b.index),
    { name: '终点', index: n - 1 },
  ]
  const out: SegmentStats[] = []
  for (let i = 0; i < bounds.length - 1; i++) {
    const s = bounds[i], e = bounds[i + 1]
    const dist = cumDist[e.index] - cumDist[s.index]
    let gain = 0, loss = 0, netSlope = 0
    if (smoothed) {
      const seg = smoothed.subarray(s.index, e.index + 1)
      ;({ gain, loss } = computeGainLoss(seg, th))
      const e0 = seg[0], e1 = seg[seg.length - 1]
      if (dist > 0 && Number.isFinite(e0) && Number.isFinite(e1)) netSlope = (e1 - e0) / dist
    }
    out.push({ fromName: s.name, toName: e.name, fromIndex: s.index, toIndex: e.index,
      dist, gain, loss, gainRate: dist > 0 ? gain / dist : 0, lossRate: dist > 0 ? loss / dist : 0, netSlope })
  }
  return out
}

/** 一键校准:求使全程 gain ≈ officialGain 的阈值(3~10m 网格搜索,返回最接近者) */
export function calibrateThreshold(track: Track, officialGain: number, opt: StatsOptions = {}): number {
  if (!track.points.ele) throw new Error('no elevation')
  const smoothed = smoothElevation(track.points.ele, opt.smoothWindow ?? 5)
  let best = 5, bestErr = Infinity
  for (let th = 3; th <= 10; th += 0.5) {
    const { gain } = computeGainLoss(smoothed, th)
    const err = Math.abs(gain - officialGain)
    if (err < bestErr) { bestErr = err; best = th }
  }
  return best
}
```

```ts
// src/core/pace/models.ts
/** Tobler 徒步函数:speed(km/h) = 6·exp(-3.5·|slope+0.05|),slope=dh/dx */
export function toblerSpeed(slope: number): number {
  return 6 * Math.exp(-3.5 * Math.abs(slope + 0.05))
}

export interface SegmentLike { dist: number; gain: number; loss: number }

export interface PaceParams {
  model?: 'tobler' | 'practical'
  flatPaceSecPerKm: number      // 平路配速 秒/公里
  vamMPerH: number              // 爬升垂直速度 米/小时
  descentFactor: number         // 下坡附加系数:每米下降折算 descentFactor 秒
  fatiguePctPerHour: number     // 每小时疲劳减速百分比
}

/** 实用档:t = 平路时间 + 爬升时间 + 下坡附加,再乘疲劳系数 */
export function practicalSegmentTime(seg: SegmentLike, p: PaceParams, elapsedSec: number): number {
  const base = (seg.dist / 1000) * p.flatPaceSecPerKm + (seg.gain / p.vamMPerH) * 3600 + seg.loss * p.descentFactor
  const fatigue = 1 + (p.fatiguePctPerHour / 100) * (elapsedSec / 3600)
  return base * fatigue
}

/** Tobler 档:按净坡度取速度,用户平路配速换算缩放系数 */
export function toblerSegmentTime(seg: SegmentLike, p: PaceParams, elapsedSec: number): number {
  const slope = seg.dist > 0 ? (seg.gain - seg.loss) / seg.dist : 0
  const userFlatKmh = 3600 / p.flatPaceSecPerKm
  const scale = userFlatKmh / toblerSpeed(0)
  const kmh = toblerSpeed(slope) * scale
  const base = (seg.dist / 1000 / kmh) * 3600
  const fatigue = 1 + (p.fatiguePctPerHour / 100) * (elapsedSec / 3600)
  return base * fatigue
}

export type WarnLevel = 'green' | 'yellow' | 'red'
export interface Arrival { etaMs: number; level: WarnLevel; marginSec: number }

/**
 * 逐段推演各 CP 到达时刻;对照关门时间:
 * red = 超关门;yellow = 余量 < 15% 段时间或 < 20 分钟;green = 其余
 */
export function estimateArrivals(
  segs: SegmentLike[], p: PaceParams, startMs: number, cutoffsMs: (number | undefined)[],
): Arrival[] {
  const out: Arrival[] = []
  let elapsed = 0
  for (let i = 0; i < segs.length; i++) {
    const segTime = (p.model === 'tobler' ? toblerSegmentTime : practicalSegmentTime)(segs[i], p, elapsed)
    elapsed += segTime
    const etaMs = startMs + elapsed * 1000
    const cutoff = cutoffsMs[i]
    let level: WarnLevel = 'green', marginSec = Infinity
    if (cutoff !== undefined) {
      marginSec = (cutoff - etaMs) / 1000
      if (marginSec < 0) level = 'red'
      else if (marginSec < Math.max(segTime * 0.15, 1200)) level = 'yellow'
    }
    out.push({ etaMs, level, marginSec })
  }
  return out
}
```
